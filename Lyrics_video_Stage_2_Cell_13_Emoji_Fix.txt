# ------------------------------------------------------------
# EXACT EMOJI ASSET INDEX
# ------------------------------------------------------------
# Twemoji filenames are based on the complete Unicode sequence:
#   💖      -> 1f496.png
#   👍      -> 1f44d.png
#   👍🏽    -> 1f44d-1f3fd.png
#   🇮🇳     -> 1f1ee-1f1f3.png
#   👩‍💻    -> 1f469-200d-1f4bb.png
#   ❤️‍🔥   -> 2764-fe0f-200d-1f525.png
#
# Build exact basename index once. No file image is opened here.
EMOJI_INDEX={}
for p in EMOJI_FILES:
    EMOJI_INDEX[Path(p).stem.lower()]=p

def emoji_codepoints(s):
    return '-'.join(f'{ord(c):x}' for c in s)

def emoji_path(ch):
    """Exact Unicode sequence -> exact Twemoji asset path."""
    if not ch:
        return None

    cp=emoji_codepoints(ch)
    p=EMOJI_INDEX.get(cp)
    if p:
        return p

    # Safe fallback ONLY for variation-selector differences.
    # Never use substring/prefix matching.
    clean=''.join(c for c in ch if ord(c)!=0xfe0f)
    if clean!=ch:
        cp2=emoji_codepoints(clean)
        p=EMOJI_INDEX.get(cp2)
        if p:
            return p

    return None

# ------------------------------------------------------------
# EMOJI-AWARE PARSING
# ------------------------------------------------------------
# Handles:
#   Jaan 💖
#   Jaan💖
#   👍🏽
#   ❤️‍🔥
#   👩‍💻
#   🇮🇳
# as complete emoji units.
EMOJI_RE=re.compile(
    r'['
    r'\U0001F1E6-\U0001F1FF'
    r'\U0001F300-\U0001FAFF'
    r'\u2600-\u27BF'
    r']'
    r'(?:\uFE0F)?'
    r'(?:[\U0001F3FB-\U0001F3FF])?'
    r'(?:\u200D'
    r'['
    r'\U0001F300-\U0001FAFF'
    r'\u2600-\u27BF'
    r']'
    r'(?:\uFE0F)?'
    r'(?:[\U0001F3FB-\U0001F3FF])?'
    r')*',
    re.UNICODE
)

def extract_emojis(text):
    return [(m.start(),m.end(),m.group()) for m in EMOJI_RE.finditer(text)]

def strip_emojis(text):
    return EMOJI_RE.sub('',text)

def attach_emojis_to_words(line):
    """
    Return plain lyric words and emoji attached to each corresponding word.

    Emoji always belongs to the nearest preceding lyric word.
    Spaces are irrelevant, so both Jaan 💖 and Jaan💖 attach 💖 to Jaan.
    """
    text=str(line or '')
    matches=extract_emojis(text)

    # Remove emoji while keeping separator space so normal words survive.
    cleaned=text
    for s,e,ch in reversed(matches):
        cleaned=cleaned[:s]+' '+cleaned[e:]

    plain_words=re.findall(r'\S+',cleaned)
    attachments=[None]*len(plain_words)

    for s,e,ch in matches:
        before=text[:s]
        prev_words=re.findall(r'\S+',strip_emojis(before))
        if not prev_words:
            continue

        target=prev_words[-1]

        # Attach to the last matching occurrence so repeated words work safely.
        for i in range(len(plain_words)-1,-1,-1):
            if plain_words[i]==target:
                if attachments[i]:
                    attachments[i]+=ch
                else:
                    attachments[i]=ch
                break

    return plain_words,attachments

    for f in FONT_FILES[:20]:
        try:
            return ImageFont.truetype(f,size)
        except:
            pass
    return ImageFont.truetype(FONT_FILES[0],size)

def draw_event(frame,ev,progress):
    st=styles[(round(ev['start'],3),ev['text'])]
    text=ev['text']
    emoji=ev.get('emoji','')

    reveal=max(
        1,
        min(
            len(text),
            int(math.ceil(len(text)*min(1,progress/.12)))
        )
    ) if text else 0

    shown=text[:reveal]
    layer=Image.new('RGBA',frame.size,(0,0,0,0))
    d=ImageDraw.Draw(layer)
    font=st['font']

    bbox=d.textbbox(
        (0,0),
        shown,
        font=font,
        stroke_width=st['sw']
    )
    tw=bbox[2]-bbox[0]

    x=max(20,min(OUTPUT_W-tw-20,st['x']))
    y=st['y']

    d.text(
        (x+3,y+3),
        shown,
        font=font,
        fill=(0,0,0,130),
        stroke_width=st['sw']+1,
        stroke_fill=(0,0,0,100)
    )
    d.text(
        (x,y),
        shown,
        font=font,
        fill=st['fill'],
        stroke_width=st['sw'],
        stroke_fill=st['stroke']
    )

    if emoji:
        # Multiple attached emoji are rendered left-to-right together.
        ep=emoji_path(emoji)
        if ep:
            try:
                with Image.open(ep) as ei:
                    ei=ei.convert('RGBA')
                    target=max(42,int(font.size*.78))
                    ei.thumbnail(
                        (target,target),
                        Image.Resampling.LANCZOS
                    )
                    ex=min(
                        OUTPUT_W-ei.width-15,
                        x+tw+10
                    )
                    ey=y+max(
                        0,
                        (font.size-ei.height)//2
                    )
                    layer.alpha_composite(ei,(ex,ey))
            except Exception:
                pass

    return Image.alpha_composite(frame.convert('RGBA'),layer)
